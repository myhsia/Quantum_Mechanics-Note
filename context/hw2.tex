\section{AQM Problem Set 2 -- 4}

\begin{problem}[Coherent-state path integral]
   Let's recall that a bosonic coherent state is defined as
  \[
    \ket|\alpha> = D(\alpha) \ket|0>, \quad
    D(\alpha) \equiv \upe^{\alpha\hat a^\dagger-\alpha^*\hat a},
  \]
  where $D(\alpha)$ is the unitary displacement operator. The
  set $\{\ket|\alpha>, \alpha \in \mathbb C\}$ forms an overcomplete basis of
  the Hilbert space
  \[
    \int \frac{\d^2\alpha}{\pi} \ketbra|\alpha><\alpha| = \identity,
  \]
  where $\d^2\alpha = \d\Re\alpha \d\Im\alpha
    = \frac{1}{2\iu}\d\bar\alpha \wedge \d\alpha$
  ($\alpha = \Re\alpha + \iu\Im\alpha$, $\bar\alpha \equiv \alpha^*$), and
  \[
    \braket<\alpha|\beta>
  = \upe^{-\frac12(\bar\alpha \alpha+\bar\beta\beta) + \bar\alpha \beta}
  \]
  We now continue to explore the use of coherent states to do path integral.
  \begin{enumext*}[label = (\alph*)]
    \item Consider a Hamiltonian $\hat H(\hat a^\dagger, \hat a; t)$ which is
    normal ordered (namely, no $\hat a^\dagger$ appears to the right of $\hat a$
    in a product), and the propagator
    \[
      \braket*\Big<\psi_f|\hat U(t_f, t_i)|\psi_i>
    = \braket*\Big<\psi_f|\mathcal T \exp\ab\Big[
        -\iu\int_{t_i}^{t_f} \d t \hat H(\hat a^\dagger, \hat a; t)]|\psi_i>
    \]
    from an initial state $\ket|\psi_i>$ to a final state $\ket|\psi_f>$, where
    $\mathcal T$ is the time-ordering operator. Derive the coherent-state path
    integral representation of the propagator by inserting resolutions of
    identity in terms of $\{\ket|\alpha>\}$
    (Hint: $\braket<\alpha|\hat H(\hat a^\dagger, \hat a; t)|\beta>
      = H(\bar\alpha, \beta; t)\braket<\alpha|\beta>$ with the normal ordering).
    Optional: compare the result with the canonical phase-space path
    integral representation.
    \item* (optional) Consider, again, the specific case of a driven
    quantum harmonic oscillator with Hamiltonian
    \[
      \hat H = \frac{\hat p^2}{2m} + \frac12 m\omega^2 \hat x^2
             + \mathcal E(t) \hat x.
    \]
    Compute the transition amplitude (as a function of time) from the
    unperturbed ground state $\ket|0>$ to an unperturbed excited state
    $\ket|n>$ by using the coherent-state path integral representation of the
    propagator derived above. By treating the driving
    term $\mathcal E(t) \hat x$ as a source term and the path integral as a
    generating functional $Z|\mathcal E(t)|$, compute the two-point correlation
    function $\braket<x(t_1)x(t_2)>$ and the four-point correlation function
    $\braket<x(t_1)x(t_2)x(t_3)x(t_4)>$.
  \end{enumext*}
\end{problem}
\begin{solution}\leavevmode
  \begin{enumext}
    \item Slicing the time range $[t_i, t_f]$ into $N$ pieces, then
    the propagator can be written as a sum (product) of small time
    \[
      \mathcal K = \braket*\Big<\psi_f|\hat U(t_f, t_i)|\psi_i>
    = \braket*\Big<\psi_f|\prod_{k=0}^{N-1}
        \exp[-\iu \epsilon \hat H(\hat a^\dagger, \hat a; t_k)]|\psi_i>
    \]
    where $\epsilon = (t_f - t_i)/N$. Inserting the identity $\identity
    = \int \frac{\d^2\alpha}{\pi} \ketbra|\alpha><\alpha|$ in the gaps between
    every two exponential terms with the symmetric
    condition $\alpha_0 = \psi_i$, $\alpha_N = \psi_f$
    \[
      \mathcal K =
      \underbrace{\int \cdots \int}_\text{$N - 1$ terms}
      \prod_{k=1}^{N-1} \frac{\d^2\alpha_k}{\pi}
      \braket<\psi_f|\upe^{-\iu\epsilon\hat H_{N-1}}|\alpha_{N-1}>
      \braket<\alpha_{N-1}| \cdots |\alpha_1>
      \braket<\alpha_1|\upe^{-\iu\epsilon\hat H_0}|\psi_i>
    \]
    Using the hint in problem, for an arbitrary overlap
    \begin{multline*}
      \braket<\alpha_k|\upe^{-\iu\epsilon\hat H_{k-1}}|\alpha_{k-1}>
    = \upe^{-\iu\epsilon H} \braket<\alpha_k|\alpha_{k-1}>\\
    = \exp\ab[-\iu\epsilon H(\bar\alpha_k, \alpha_{k-1}, t_k)
      -\frac12(\bar\alpha_k \alpha_k + \bar\alpha_{k-1} \alpha_{k-1})
      +\bar\alpha_k \alpha_{k-1}]
    \end{multline*}
    To calculate the products of overlaps, sum the exponents first.
    \begin{enumext}
      \item The second term.
      \[
        -\frac12\sum_{k=1}^N (\bar\alpha_k \alpha_k
          + \bar\alpha_{k-1}\alpha_{k-1})
      = -\frac12(|\alpha_N|^2 + |\alpha_0|^2)
      - \sum_{k=1}^{N-1} |\alpha_k|^2
      \]
      \item The third term.
      \[
        \sum_{k=1}^N \bar\alpha_k \alpha_{k-1}
      = \sum_{k=1}^N \bar\alpha_k (\alpha_k - \alpha_k + \alpha_{k-1})
      = \sum_{k=1}^N |\alpha_k|^2 - \sum_{k=1}^N (\alpha_k - \alpha_{k-1}).
      \]
    \end{enumext}
    Combining the first, the second, and the third term, one will obtain the sum
    \begin{align*}
      S_\text{overlap} & = -\frac12(|\alpha_N|^2 + |\alpha_0|^2)
        - \sum_{k=1}^{N-1} |\alpha_k|^2 + \sum_{k=1}^N |\alpha_k|^2
        - \sum_{k=1}^N \bar\alpha_k (\alpha_k - \alpha_{k-1})
        - \iu \sum_{k=1}^N H(\bar\alpha_k, \alpha_{k-1}, t_k) \epsilon\\
    & = \frac12(|\alpha_N|^2 - |\alpha_0|^2)
        - \sum_{k=1}^N \epsilon \dot\alpha_k
        - \iu \sum_{k=1}^N H(\bar\alpha_k, \alpha_{k-1}, t_k) \epsilon,
    \end{align*}
    where the first-order Taylor
    approximation $\alpha_k - \alpha_{k-1} = \epsilon \dot\alpha$ is used.
    In the continuum limit, $\epsilon \to \d t$, the sum can be written
    as integral
    \begin{multline*}
      S_\text{overlap} = \frac12(|\alpha_N|^2 - |\alpha_0|^2)
        - \int_{t_i}^{t_f} \d t \bar\alpha \dot\alpha
        - \iu \int_{t_i}^{t_f} \d t H(\bar\alpha, \alpha, t)\\
      = \frac12(|\alpha_N|^2 - |\alpha_0|^2) + \iu \int_{t_i}^{t_f} \d t
          [\iu \bar\alpha \dot\alpha - H(\bar\alpha, \alpha, t)]
    \end{multline*}
    Substituting it into the propagator
    \[
      \mathcal K = \int_{\alpha(t_i)=\alpha_0}^{\alpha(t_f)=\alpha_N}
      \mathcal D^2[\alpha]
      \exp\ab\{\frac12(|\alpha_N|^2 - |\alpha_0|^2) + \iu \int_{t_i}^{t_f} \d t
            [\iu \bar\alpha \dot\alpha - H(\bar\alpha, \alpha, t)]\},
    \]
    where $\mathcal D^2[\alpha]$ denotes the
    measure $\prod_{k=1}^{N-1} \frac{\d^2\alpha_k}{\pi}$ in the continuum limit.

    For the canonical phase-space PI rep., taking $\alpha = x + \iu p$, then
    $\alpha \dot\alpha = x\dot x + p\dot p - x\dot p + p\dot x$,
    substituting into $\iu\bar\alpha \dot\alpha - H$ will reproduce the
    Lagrangian $p\dot x - H \equiv \mathcal L$, and so yields the usual
    phase-space PI.
    \item Using the creation and annihilation operators to rewrite
    the Hamiltonian
    \[
      \hat H = \omega\ab(\hat a^\dagger \hat a + \frac12)
        + \lambda \mathcal E(t) (\hat a + \hat a^\dagger),
      \qq{where} \lambda = \sqrt{\frac1{2m\omega}}.
    \]
  \end{enumext}
\end{solution}

\begin{problem}[Discrete translational symmetry]
  Consider the following Hamiltonian for a 1D system
  \[
    \hat H = \frac{\hbar^2}{2m}\hat k^2 + V\cos\ab(\frac{2\pi\hat x}\lambda),
  \]
  where $\lambda$ is a positive real number.
  \begin{enumext*}[label = (\alph*)]
    \item Compute the commutator $[\hat k, \hat H]$. Is the eigenvalue
    of $\hat k$ a good quantum number in this problem?
    \item Define $\hat T_\lambda = \upe^{-\iu\lambda\hat k}$, show that
    $\hat T_\lambda\ket|x>$ is an eigenstate of $\hat x$. What is the physical
    meaning of $\hat T_\lambda$ as a transformation of quantum states? Compute
    the comtator $[\hat T_\lambda, \hat H]$. Is the eigenvalue
    of $\hat T_\lambda$ a good quantum number in this problem?
    \item Find all engenstates of $\hat T_\lambda$ with eigenvalue $1$ and $-1$,
    respectively. Write down the projected Hamiltonian in these two
    fixed $\hat T_\lambda$-eigenvalue sub-Hilbert-spaces, respectively.
    \item* (optional) We will call the $\hat T_\lambda$-eigenvalue $1$
    (resp., $-1$) subspace the $k = 0$ (resp., $k = \pi$) subspace. To the
    second order in $V$ as perturbation, compute the approximate energy and the
    corresponding wave-function of the lowest-energy eigenstate of the
    Hamiltonian in the $k = 0$ space, and compute the first excitation gap
    (i.e. the energy difference between the lowest and the second-lowest energy
    eigenstates of the Hamiltonian) in the $k = \pi$ space.
  \end{enumext*}
\end{problem}
\begin{solution}\leavevmode
  \begin{enumext}
    \item Since the commutator cam be computed using the identity
    \[
      [a, b^n] = [\hat a, \hat b]\hat b^{n-1}
                + \hat b[\hat a, \hat b]\hat b^{n-2} + \cdots
                + \hat b^{n-1}[\hat a, \hat b]
    = \sum_{m=0}^{n-1} a^m [a, b] b^{n-1-m}.
    \]
    Given $[\hat k, \hat x] = -\iu$, then
    \[
      [\hat k, \hat x^n] = (-\iu) \sum_{m=0}^{n-1} \hat x^{n-1}
    = (-\iu) n \hat x^{n-1}.
    \]
    Consider an arbitrary function can be expanded into a power series
    $f(\hat x) = \sum_{n=0}^\infty a_n \hat x^n$.
    Then, the commutator of $\hat k$ and it becomes
    \[
      [\hat k, f(\hat x)] = \sum_{n=1}^\infty a_n [\hat k, \hat x^n]
    = (-\iu) \sum_{n=1}^\infty a_n n \hat x^{n-1}
    = (-\iu) f'(\hat x).
    \]
    Taking $f(\hat x) = \cos(2\pi\hat x/\lambda)$.
    Since $[\hat k, \hat k^2] = 0$, then
    \[
      [\hat k, \hat H]
    = V (-\iu) (-\frac{2\pi}{\lambda}) \sin\ab(\frac{2\pi\hat x}{\lambda})
    = \frac{2\pi\iu V}{\lambda} \sin\ab(\frac{2\pi\hat x}{\lambda}).
    \]
    Since $[\hat k, \hat H] \neq 0$, so $\hat k$'s eigenvalue is not a good
    quantum number.
    \item Consider $\hat T_\lambda \hat x \hat T_\lambda^{-1}$, due to the
    BCH identity
    \[
      \hat T_\lambda \hat x \hat T_\lambda^{-1}
    = \upe^{-\iu\lambda\hat k} \hat x \upe^{\iu\lambda\hat k}
    = x + [x, -\iu\lambda \hat k] \xlongequal{[x, k] = \iu} \hat x + \lambda.
    \]
    Rearrange it, then
    \[
      \hat T_\lambda \hat x = (\hat x + \lambda) \hat T_\lambda
    \]
    Act the both sides on $\ket|x>$
    \begin{align*}
      \text{LHS} & = \hat T_\lambda \hat x \ket|x> = \hat T_\lambda x \ket|x>
    = x \hat T_\lambda \ket|x> = x \ket|x + \lambda>,\\
      \text{RHS} & = (\hat x + \lambda) \hat T_\lambda \ket|x>
    = (\hat x + \lambda) \ket|x + \lambda>
    \end{align*}
    Rearrange it, then
    \[
      (\hat x + \lambda) \ket|x + \lambda> = x\ket|x + \lambda>, \quad
      \hat x \ket|x + \lambda> = (x - \lambda) \ket|x + \lambda>.
    \]
    This is the eigenequation for $\hat x$ with the eigenvalue $x - \lambda$.
    The physical meaning is that $\hat T_\lambda$ is the
    operator that translates states by distance $\lambda$.

    Concerning the commutator $[\hat T_\lambda, H]$.
    It is obviously that $\hat T_\lambda$ is commutative to the kinetic term,
    then due to the BCH identity
    \[
      \hat T_\lambda \ab(\frac{\hbar^2}{2m} \hat k^2) \hat T_\lambda^{-1}
    = \frac{\hbar^2}{2m} \hat k^2.
    \]
    For the potential term, consider
    \[
      \hat T \hat x \hat T^{-1} \xlongequal{\text{BCH identity}}
      \hat x - [-\iu\lambda \hat k, \hat x] = \hat x + \lambda.
    \]
    Since the potential term can be expanded in terms of a combination of
    different power of $\hat x$, and $\hat T^{-1}\hat T$ can be inserted between
    every two $\hat x$, which shifts them to $\hat x + \lambda$. Hence
    \[
      \hat T \ab[V\cos\ab(\frac{2\pi\hat x}{\lambda})] \hat T^{-1}
    = V\cos\ab[\frac{2\pi(\hat x + \lambda)}{\lambda}]
    = V\cos\ab(\frac{2\pi\hat x}{\lambda}).
    \]
    The same as the potential term.
    Since the commutator can be written as
    \[
      [\hat T_\lambda, \hat H]
    = \hat T_\lambda \hat H \hat T_\lambda^{-1} \hat T_\lambda
      - \hat H \hat T_\lambda
    = (\hat T_\lambda \hat H \hat T_\lambda^{-1} - \hat H) \hat T_\lambda.
    \]
    We can obtain $\hat T_\lambda \hat H \hat T_\lambda^{-1} = \hat H$.
    So, $[\hat T_\lambda, \hat H] = 0$, the eigenvalue of $\hat T_\lambda$
    is a good quantum number.
    \item 
  \end{enumext}
  \begin{enumext}
    \item Eigenstates of $+1$
    \[
      \{\ket|2n> \equiv \ket|k = \frac{2n\pi}{\lambda}>, n \in \mathbb Z\},
    \]
    Eigenstates of $-1$
    \[
      \{\ket|2n + 1> \equiv \ket|k = \frac{(2n + 1)\pi}{\lambda}>, n \in \mathbb Z\}.
    \]
    The kinetic term
    \[
      \braket<n|\frac{\hbar^2\hat k^2}{2m}|n'> = \ab(\frac{n\pi}{\lambda})^2
      \frac{\hbar^2}{2m} \delta_{nn'} = E_n \delta_{nn'}
    \]
    and the move operator in $k$-space
    \[
      \frac12\braket<|\upe^{\iu\frac{2\pi}{\lambda}\hat x}+\upe^{-\iu\frac{2\pi}{\lambda}\hat x}|2n'>
    = \frac12(\braket<2n|2(n' + 1)> + \braket<2n'|2(n' - 1)>)
    \]
    \item $k = 0$, $E_g = E_0 - 2\frac{(V_{1,2})^2}{E_2 - E_0}$
    \[
      \ket|g> =
      \ab(1 - \frac{V^2}{2E_2^2}) \ket|0>
    - \frac{V}{2E_0} (\ket|-2> + \ket|+2>)
    + \frac{V^2}{4E_4E_2} (\ket|-4> + \ket|+4>).
    \]
    When $k = \pi$, $\Delta = V$.
  \end{enumext}
\end{solution}

\begin{problem}[Representation of $D_3$]
  Let's extend the example used in our classes by including two
  flavors ($s = \pm$) on each vertices of the equilateral triangle, such that
  the Hilbert space becomes $V \equiv \operatorname{span}
    (\{\ket|i = 9,1,2; s = \pm>\})$. The action of the two generators of the
  dihedral group $D_3 = \braket<R,\rho|R^3 = \rho^2 = (\rho R)^2 = e>$ on this
  Hilbert space is given by
  \[
    \hat R\ket|i, s> = \ket|i + 1, s>, \quad
    \hat \rho \ket|i, s> = \ket|-i, -s>.
  \]
  Here, $\hat R$ and $\hat \rho$ are unitary operators, and integers such as
  $i - 1$ and $-i$ are understood to be modulo $3$.
  \begin{enumext}
    \item Check that as operators, $\hat R^3 = \hat\rho^2
    = (\hat \rho\hat R)^2 = \identity$, such that they indeed generate a
    representation of $D_3$.
    \item Derive the most general form of the Hamiltonian $\hat H$ on $V$ which
    is invariant under the action of $D_3$, i.e., $[\hat H, \hat R]
    = [\hat H, \rho] = 0$. How many independent (real) parameters does it have?
    \item Consider the following Hamiltonian
    \[
      \hat H = t(\hat R_+ \oplus \hat R_-^2)
             + t^*(\hat R_+^2 \oplus \hat R_-)
             + s(\hat \rho_0 + \hat \rho_1 + \hat \rho_2),
    \]
    where $\hat R_{s=\pm}$ is the restriction of $\hat R$ on the
    subspace $V_s \equiv \operatorname{span}(\{\ket|i = 0,1,2;s>\})$
    (such that $\hat R = \hat R_+ \oplus \hat R_-$);
    $\hat \rho_{i=0,1,2} \equiv \hat\rho \hat R^i$ is the reflection with
    respect to the $i$-th vertex; $t$ is a complex parameter and $s$ is a real
    parameter. Check that this Hamiltonian is invariant under the action
    of $D_3$. Further check that the three parts of the Hamiltonian: $\hat H_R
    \equiv t(\hat R_+ \oplus \hat R_-^2)$, $\hat H_R^\dagger
    = t^*(\hat R_+^2 \oplus \hat R_-)$
    and $\hat H_\rho \equiv s(\hat \rho_0 + \hat \rho_1 + \hat \rho_2)$ all
    commute with each other: $[\hat H_R, \hat H_R^\dagger] = 0$,
    $[\hat H_R, \hat H_\rho] = 0$, and $[\hat H_R^\dagger, \hat H_\rho] = 0$.
    Use this preparation to solve the eigensystem of $\hat H$ exactly. Analyze
    the result based on the irreps of $D_3$.
  \end{enumext}
\end{problem}
\begin{solution}\leavevmode
  \begin{enumext}
    \item $(\hat\rho \hat R) \ket|i,s> = \ket|-i - 1, s>$
    \item
    \[
      \hat R\ketbra|i,s><i',s'|\hat R^{-1}
    = \ketbra|i + 1, s><i' + 1, s'|, \quad
      \hat\rho \ketbra|i, s><i', s'| \hat\rho^{-1}
    = \ketbra|-i, -s><-i', -s'|.
    \]
    \[
      \begin{pNiceArray}[first-col]{ccc:ccc}
        \ket|0+> & \epsilon & a & b & c & d & e\\
        \ket|1+> & b & \epsilon & a & e & c & d\\
        \ket|2+> & a & b & \epsilon & d & e & c\\
        \hdottedline
        \ket|0-> & c & e & d & \epsilon & b & a\\
        \ket|1-> & d & c & e & a & \epsilon & b\\
        \ket|2-> & e & d & c & b & a & c
      \end{pNiceArray}
    \]
    Then, there are $\epsilon$, $c$, $d$, $e \in \mathbb R$ ($+$),
    $b = a^* \in \mathbb C$ ($-$).
    \item $\hat \rho \hat R_\pm \hat \rho^{-1} = \hat R^2_\mp$,
    then
    \[
      \hat \rho (\hat R_+ + \hat R_-^2) \hat \rho^{-1}
    = \hat R_-^2 + \hat R_+^2.
    \]
    $\hat R \hat \rho_i \hat R^{-1} = \hat \rho_{i+1}$,
    then
    \[
      \hat R(\sum_i \hat \rho_i) \hat R^{-1} = \sum_i \hat \rho_i.
    \]
    The commutator $[\hat H_R, \hat H_\rho] = 0$,
    $\hat \rho_i = \hat \rho \hat R^2$.

    Consider $\hat R_s$
    \[
      \hat R_s \ket|k, s> = \omega^k \ket|k, s>, \quad
      \omega = \upe^{\iu\frac{2\pi}{3}}, \quad
      \ket|k, s> = \frac1{\sqrt3} \sum_i \omega^{-ki} \ket|i, s>
    \]
    The eigenvalues for $\hat R_\pm$
    \begin{align*}
      \omega^0, & \{\ket|k = 0, +>, \ket|k = 0, ->\}\\
      \omega^1, & \{\ket|k = 1, +>, \ket|k = 1, ->\}\\
      \omega^2, & \{\ket|k = 2, +>, \ket|k = 2, ->\}.
    \end{align*}
    $\hat R_-\ket|k, +> = 0$.
    
    Advanced.
    \[
      \hat \rho_i \ket|k,s>
    = \hat \rho \hat R^i \ket|k, s>
    \xlongequal{\hat R = \hat R_+ + \hat R_-}
    \omega^{ki} \ket|-k, -s>.
    \]
    \[
      \hat H_\rho \ket|k,s>
    = \sum_i \hat \rho_i \ket|k, s>
    = \sum_i \omega^{k_i} \ket|-k, -s> = 3 \delta_{k,0} \ket|-k, -s>
    \]
    \begin{center}
      \begin{tabular}{c*6c}
        \toprule
        States & $\ket|k = 0,+>$ & $\ket|k = 0,->$
               & $\ket|k = 1,+>$ & $\ket|k = 1,->$
               & $\ket|k = 2,+>$ & $\ket|k = 2,->$\\
        \midrule
        $\hat H_R \sim \hat R_+ + \hat R_-^2$ &
        $\omega^0$ & $\omega^0$ & $\omega^1$ &
        $\omega^2$ & $\omega^2$ & $\omega^1$\\
        $\hat H_R \sim \hat R_- + \hat R_+^2$ &
        $\omega^0$ & $\omega^0$ & $\omega^2$ &
        $\omega^1$ & $\omega^1$ & $\omega^2$\\
        \bottomrule
      \end{tabular}
    \end{center}
    \begin{enumext}
      \item For the states $\{\ket|k = 0,+>, \ket|k = 0,->\}$, their eigenvalues
      $t\omega + (t\omega)^*$, $H_\rho = 3\begin{psmallmatrix}
        & & 1\\
        1
      \end{psmallmatrix}$.
      \item For the states $\{\ket|k = 1,+>, \ket|k = 2,->\}$ and
      $\{\ket|k = 1,->, \ket|k = 2,+>\}$ their eigenvalues
      $t\omega^* + t^*\omega$, $H_\rho = \begin{psmallmatrix}
        0
      \end{psmallmatrix}$.
    \end{enumext}
  \end{enumext}
\end{solution}

\begin{problem}[Two-site Hubbard model]
  Consider the Fock space for two sites and two spin-$1/2$ states on each site,
  generated by $\{c_{1\uparrow}^\dagger, c_{1\downarrow}^\dagger,
  c_{2\uparrow}^\dagger, c_{2\downarrow}^\dagger\}$.
  \begin{enumext*}[label = (\alph*)]
    \item What is the dimension of this Fock space? Write down the basis for
    this Fock space grouped by the total particle number.
    \item Compute the full spectrum of the following Hamiltonian (assuming all
    parameters to be real)
    \[
      H = \sum_{\substack{i=1,2\\s=\uparrow,\downarrow}}
          \epsilon c_{is}^\dagger c_{is}
        - \sum_{s=\uparrow,\downarrow}(tc_{1s}^\dagger c_{2s} + \text{H.c.})
        + \sum_{i=1,2} Uc_{i\uparrow}^\dagger c_{i\uparrow}
                        c_{i\downarrow}^\dagger c_{i\downarrow}.
    \]
    (Hint: Find conserved particle numbers and compute in each
    fixed-particle-number sector separately.)
    \item Further assuming $\epsilon < 0$ and $U \gg (-\epsilon) > t > 0$, find
    the ground state and the first excited state(s) of the Hamiltonian, and
    compute the entangelement entropy between site 1 and site 2 for each of
    these states.
    \item* (optional) In the subspace spanned by
    $(\hat c_{2\uparrow}^\dagger   \hat c_{1\uparrow}^\dagger,
      \hat c_{2\downarrow}^\dagger \hat c_{1\downarrow}^\dagger,
      \hat c_{2\downarrow}^\dagger \hat d_{1\uparrow}^\dagger,
      \hat c_{2\uparrow}^\dagger   \hat c_{1\downarrow}^\dagger) \ket|0>$,
    compute the spectrum and find the ground state (for both $J > 0$ and $J < 0$
    cases) of the following Hamiltonian
    \[
      \hat H_\text{ex} = \sum_{\alpha=x,y,z} J
        (\bm c_1^\dagger \hat \sigma_\alpha \bm c_1)
        (\bm c_2^\dagger \hat \sigma_\alpha \bm c_2),
    \]
    where $\bm c_{i=1,2}^\dagger \equiv (c_{i\uparrow}^\dagger,
    c_{i,\downarrow}^\dagger)$, and $\sigma_{\alpha=x,y,z}$ are Pauli matrices.
    Compare these results with those obtained in part (c).
  \end{enumext*}
\end{problem}
\begin{solution}
  $N_\uparrow = \sum_{i = 1,2} n\uparrow$ and $N_\downarrow$ are commute with
  the Hamiltonian, so they are conserved.
  \begin{enumext}
    \item 0, 0, dim 1, $\ket|0,0>$ $H_\text{sub} = 0$
    \item $N_\text{tot} = 1$, $H_\text{sub} = \epsilon + \begin{psmallmatrix}
      & -t\\-t
    \end{psmallmatrix}$
    \begin{enumext}
      \item 1, 0, dim 2, $\ket|0,\uparrow>$, $\ket|\uparrow,0>$
      \item 0, 1, dim 2
    \end{enumext}
    \item $N_\text{tot} = 2$, 
    \begin{enumext}
      \item 2, 0, dim 1, $\ket|\uparrow, \uparrow>$, $H_\text{sub} = 2\epsilon$
      \item 1, 1, dim 4
      \item 0, 2, dim 1,
      $\ket|\downarrow, \downarrow>$, $H_\text{sub} = 2\epsilon$
    \end{enumext}
    \item $N_\text{tot} = 3$
    \begin{enumext}
      \item 2, 1, dim 2, $\ket|\uparrow, \uparrow\downarrow>$,
      \item $\ket|\uparrow\downarrow, \uparrow>$,
      $H_\text{sub} = U + 3\epsilon +
      \begin{psmallmatrix}
        & t\\t
      \end{psmallmatrix}$
      \item 1, 2, dim 2
    \end{enumext}
    \item $N_\text{tot} = 4$, 2, 2, dim 1,
    $\ket|\uparrow\downarrow, \uparrow\downarrow>$, $4\epsilon + 2U$.
    \[
      \begin{pNiceArray}[first-col, last-row]{cc:cc}
        \ket|0,\uparrow\downarrow>  & U & 0 & -t & t\\
        \ket|\uparrow\downarrow, 0> & 0 & U & -t & t\\
        \hdottedline
        \ket|\uparrow, \downarrow> & -t & -t & 0 & 0\\
        \ket|\downarrow, \uparrow> & t  & t  & 0 & 0\\
        & \bra<0,\uparrow\downarrow| & \bra<\uparrow\downarrow,0| 
        & \bra<\uparrow, \downarrow| & \bra<\downarrow, \uparrow| 
      \end{pNiceArray} + 2\epsilon
    \]
  \end{enumext}
  The total dimension is $16$.
\end{solution}
